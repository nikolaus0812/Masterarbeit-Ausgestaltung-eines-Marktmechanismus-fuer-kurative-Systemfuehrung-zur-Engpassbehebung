\noindent\textbf{Absatz 1, was Kapitel~\ref{ch:results} geliefert hat.}
\begin{itemize}
    \item Kapitel~\ref{ch:results} liefert für jede Stunde des Jahres 2025 je Richtung einen kurativen Reservierungspreis
    \item Der Median beträgt 12,22\,\EurMWh{} entladend und 9,85\,\EurMWh{} ladend, das arithmetische Mittel 23,64 und 24,33\,\EurMWh{}
    \item Die Verteilung ist in beiden Richtungen rechtsschief, und die Maxima erreichen 1008 und 510\,\EurMWh{}
    \item Über den Tag schwankt der Median der Summe beider Richtungen um den Faktor 8,3 und über die Monate um 4,2
    \item Die Zahlung des \ac{ÜNB} bei vollständiger Bindung beträgt 420,0\,\TsdEurMWa{} gegen einen Referenzerlös von 340,6\,\TsdEurMWa{}
    \item Auf die \ac{aFRR} entfallen 76,7 Prozent des Bruttoerlöses, sodass die kurative Reservierung vor allem Regelleistung verdrängt
\end{itemize}

\noindent\textbf{Absatz 2, woran dieses Ergebnis hängt.}
% Die ausfuehrliche Fassung des Schlussabsatzes von 4.6.
\begin{itemize}
    \item Übergang, vier Sensitivitäten sagen, welche Annahme das Ergebnis trägt
    \item Die Annahmen über die \ac{aFRR} tragen es am stärksten, denn ohne die \ac{aFRR} fällt der Median auf 28 Prozent entladend und 43 Prozent ladend
    \item Unter freier Lieferung der \ac{aFRR}-Arbeit steigt der Median um 54 und 110 Prozent, was für eine einzelne Anlage von 100\,\si{MW} nicht zu erwarten ist

    \item Eine Viertelung der vorgehaltenen Energie je Abruf senkt ihn dagegen nur um 4,0 und 12,7 Prozent
    \item Die Maxima reagieren auf jede Annahme stärker als der Median, sodass die Spitze der Verteilung weniger belastbar ist als ihr Hauptfeld

\section{Einordnung des Ergebnisses}
\label{sec:result_classification}
% Ziel 2 Seiten, vier Absaetze.
% Fragen des Abschnitts: Was ist die ermittelte Zahl und was ist sie nicht?
% Wie hoch liegt sie und wie streut sie? Gegen welchen Markt ist sie ein
% Preis? Wie verhaelt sie sich zu den Kosten des Engpassmanagements?

\noindent\textbf{Absatz 1, der ermittelte Preis ist die Untergrenze eines Gebots.}
\begin{itemize}
    \item Anschluss an Kapitel~\ref{ch:results}, das für jede Stunde des Jahres 2025 je Richtung einen kurativen Reservierungspreis liefert
    \item Der Preis misst allein die entgangene Marktopportunität derselben Stunde und folgt damit dem Indifferenzprinzip aus Abschnitt~\ref{sec:redispatch_compensation}
    \item Nicht enthalten sind die Wahrscheinlichkeit eines Abrufs und die Sanktion bei Nichterfüllung, denn das Optimierungsmodell bildet beide nicht ab
    \item Ebenso wenig enthalten ist der Aufwand, den ein Betreiber für das Anbieten selbst trägt
    \item Ein Betreiber böte deshalb über dem ermittelten Preis, und zwar um einen Aufschlag, den diese Arbeit nicht beziffert
    % Kernsatz des Absatzes, eigenstaendige Ableitung.
    \item Der ermittelte kurative Reservierungspreis ist damit die Untergrenze eines Gebots und nicht das Gebot
\end{itemize}

\noindent\textbf{Absatz 2, eine einzelne Zahl gibt den Preis nicht wieder.}
\begin{itemize}
    \item Übergang, wie hoch diese Untergrenze liegt, hängt an der Stunde
    \item Der Median beträgt 12,22\,\EurMWh{} in der Entlade- und 9,85\,\EurMWh{} in der Laderichtung, das arithmetische Mittel 23,64 und 24,33\,\EurMWh{}
    \item Der Abstand beider Maße misst die Rechtsschiefe, nämlich das 1,93-Fache entladend und das 2,47-Fache ladend
    \item Die Hälfte der Stunden ist zum Median oder darunter zu reservieren, während die Maxima bei 1008 und 510\,\EurMWh{} liegen
    \item Für die Beschaffung folgt daraus, dass der \ac{ÜNB} den größten Teil der Vorhaltung günstig bekommt und allein für die Spitze viel zahlt
    % Vorgabe des Verfassers vom 23.09.2026, HANDOFF Abschnitt 7.
    \item Die teuren Stunden sind zugleich die am schlechtesten vorhersagbaren, denn sie hängen an einzelnen Preisspannen des Energiemarkts
\end{itemize}

\noindent\textbf{Absatz 3, der kurative Reservierungspreis ist ein Preis gegen die Regelleistung.}
\begin{itemize}
    \item Übergang, gegen welche Vermarktung der Preis entsteht, zeigt die Aufteilung des Erlöses
    \item Auf die \ac{aFRR} entfallen mit Leistung und Arbeit 76,7 Prozent des Bruttoerlöses der Bezugsanlage
    \item Der Leistungspreis der \ac{aFRR} entfällt mit der reservierten Stunde vollständig, weil er an eine Zeitscheibe von vier Stunden gebunden ist
    \item Die Arbitrage an den Energiemärkten lässt sich dagegen zwischen den Stunden eines Tages verschieben, solange der Tag eine Staffel gleichwertiger Spannen trägt
    \item Der Energiemarkt setzt den Preis deshalb nur an jedem zwanzigsten Tag des Jahres, an dem diese Staffel einbricht
    % Kernsatz, eigenstaendige Ableitung aus Kapitel 4.
    \item Der kurative Reservierungspreis ist damit im Kern ein Preis gegen die Regelleistung, und wer ihn senken will, muss an der Regelleistung ansetzen
\end{itemize}

\noindent\textbf{Absatz 4, das Verhältnis zu den Kosten des präventiven Redispatch.}
\begin{itemize}
    \item Übergang, ob dieser Preis günstig ist, bemisst sich am bestehenden Engpassmanagement
    \item Die Bundesnetzagentur weist für das Jahr 2025 einen Arbeitspreis von 101 Euro je bewegter Megawattstunde aus \cite{bundesnetzagentur_smard_engpassmanagement_2026}
    \item Beide Größen tragen dieselbe Dimension und meinen Verschiedenes, denn die eine vergütet Vorhaltung und die andere bewegte Arbeit
    \item Das Tagesmittel des kurativen Reservierungspreises bleibt an 362 von 365 Tagen entladend und an 364 von 365 Tagen ladend unter dieser Linie
    \item Die kurative Vorhaltung ist damit über das ganze Jahr einplanbar
    \item Ob sie im Einzelfall günstiger ist als die Maßnahme, die sie ersetzt, hängt an der Zahl der vorgehaltenen Stunden je Abruf, die diese Arbeit nicht bestimmt
\end{itemize}

\section{Einordnung in den regulatorischen Rahmen}
\label{sec:regulatory_discussion}
% Ziel 2,5 Seiten, fuenf Absaetze. Material B1 bis B7, F7, F8, G1.
% Fragen des Abschnitts: Kennt der geltende Rahmen die Groesse, die hier
% bepreist wird? Traegt die Verguetungsstruktur der Festlegung sie? Ist die
% Luecke fuer Speicher dokumentiert? Gibt es eine bestehende
% Beschaffungsform, an die der Mechanismus anschliessen kann?

\noindent\textbf{Absatz 1, der geltende Rahmen kennt keine Vergütung der Vorhaltung.}
\begin{itemize}
    \item Übergang aus Abschnitt~\ref{sec:result_classification}, der ermittelte Preis vergütet Vorhaltung, und der geltende Rahmen kennt diese Größe nicht
    \item §\,13a Abs.\,2 \ac{EnWG} entschädigt den durchgeführten Eingriff und schreibt eine kostenbasierte Abrechnung vor \cite{bundesministerium_der_justiz_energiewirtschaftsgesetz_2025}
    \item Eine marktbasierte Beschaffung des Redispatch ist damit im geltenden Recht ausgeschlossen
    \item Redispatch~2.0 ist ein Duldungs- und Entschädigungsregime, in dem der Anlagenbetreiber die Anweisung hinnimmt und dafür schadlos gestellt wird \cite{bundesnetzagentur_festlegung_2024}
    \item Die kurative Reservierung setzt dagegen ein Gebot des Betreibers voraus, denn ohne Gebot kommt die Maßnahme nicht zustande
    \item Der entworfene kurative Marktmechanismus ist innerhalb des geltenden Rahmens deshalb nicht abbildbar und setzt eine Änderung des §\,13a \ac{EnWG} voraus
\end{itemize}

\noindent\textbf{Absatz 2, die Vergütungsstruktur der Festlegung erfasst den ermittelten Preis nicht.}
\begin{itemize}
    \item Übergang, auch dort, wo der Rahmen eine Vergütung vorsieht, trifft ihre Struktur den ermittelten Preis nicht
    \item Die Festlegung bildet die Vergütung als Erzeugungsauslagen zuzüglich des Maximums aus anteiligem Werteverbrauch und Opportunität \cite{bundesnetzagentur_festlegung_2024}
    \item Die Struktur ist damit ein Auswahlverhältnis und keine Summe
    \item Ist der Werteverbrauch die größere der beiden Größen, geht die Opportunität in die Zahlung gar nicht ein
    \item Der in Kapitel~\ref{ch:results} ermittelte Preis ist gerade die Opportunität und würde unter dieser Struktur nicht zahlungswirksam
    \item Eine Vergütung der kurativen Vorhaltung verlangt deshalb eine eigene Struktur und nicht die Anwendung der bestehenden
\end{itemize}

\noindent\textbf{Absatz 3, die Methode der Festlegung passt zum Betrieb eines Speichers nicht.}
% F7 und F8 des alten Geruests. Beide kritisieren das Referenzverfahren und
% nicht das eigene Modell, deshalb stehen sie hier und nicht in 5.5.
\begin{itemize}
    \item Übergang, neben der Struktur ist die Bemessung selbst zu prüfen
    \item Das Volatilitätsmaß der Festlegung ist ein Maß der Verteilungsränder, denn 69 Prozent der Jahressumme stammen aus dem größten Prozent der Viertelstunden
    \item Eine Größe mit dieser Konzentration als Maßstab einer Vergütung zu verwenden, ist begründungsbedürftig
    \item Das Referenzverfahren bewertet einzelne Zeitscheiben zudem unabhängig voneinander
    \item Für die Bezugsanlage mit 2,5 Stunden Energieinhalt je Leistung ist die Kopplung über den Ladezustand jedoch bindend
    \item Beide Eigenschaften zusammen sprechen dafür, den Wert der Bindung aus dem Betrieb der Anlage zu bestimmen statt aus einer Preisreihe
\end{itemize}

\noindent\textbf{Absatz 4, der Regelungsbedarf für Speicher ist ein dokumentierter offener Punkt.}
\begin{itemize}
    \item Übergang, die Behörde hat die Anwendung auf Speicheranlagen selbst offengelassen
    \item Anlage~1 der Festlegung hält die Vorgaben für \ac{PSKW} auf \ac{BESS} auch ohne weitere Anpassungen für anwendbar \cite{bundesnetzagentur_festlegung_2024}
    \item Zugleich hat die Beschlusskammer die Behandlung von \ac{BESS} einem späteren Hinweis vorbehalten
    \item Die Regelungslücke ist damit dokumentiert und keine Behauptung dieser Arbeit
    \item Für Anlagen zur Erzeugung aus erneuerbaren Energien bestehen eigene Vorgaben, sodass eine technologiespezifische Regelung dem Rahmen nicht fremd ist \cite{iaew_abschlussbericht_2023, iaew_folgestudie_2024}
    \item Das Ergebnis dieser Arbeit liefert für diesen offenen Punkt eine Größenordnung der Vorhaltekosten eines \ac{BESS}
\end{itemize}

\noindent\textbf{Absatz 5, die Nähe zum kapazitätsbasierten Redispatch.}
% NEU EINZUFUEHRENDER BEGRIFF. Der kapazitaetsbasierte Redispatch ist in den
% Kapiteln 1 bis 4 nicht eingefuehrt, weil der vergleichende Unterabschnitt
% am 22.09.2026 geparkt wurde. Er ist hier in einem Satz zu definieren.
% BELEG NOCH ZU PRUEFEN, in Betracht kommt consentec_ausarbeitung_2024.
\begin{itemize}
    \item Übergang, eine bestehende Beschaffungsform kommt der kurativen Vorhaltung nahe
    \item Der kapazitätsbasierte Redispatch beschafft Leistung, die für einen späteren Eingriff bereitgehalten wird, und vergütet diese Bereithaltung
    \item Er beschafft damit im Kern dieselbe Größe wie die kurative Vorhaltung
    \item Der Unterschied liegt im Auslöser, denn dort löst die Vorschaurechnung des \ac{ÜNB} aus und hier der Eintritt des Fehlerfalls
    \item Der zweite Unterschied liegt in der Reaktionszeit, die dort im Bereich von Stunden und hier in der Klasse von etwa zwei Minuten liegt
    \item Für die Einführung folgt daraus, dass der kurative Marktmechanismus an eine bestehende Beschaffungsform anschließen kann statt eine neue zu begründen
\end{itemize}

\section{Bewertung anhand des Anforderungskatalogs}
\label{sec:requirements_assessment}
% Ziel 2 Seiten, vier Absaetze. Material A1 bis A8 aus 3.1.1, dazu C1 und D5.
% Fragen des Abschnitts: Welche der acht Anforderungen erfuellt der Entwurf,

\noindent\textbf{Absatz 1, vier Anforderungen betreffen den Zuschnitt und sind im Entwurf umgesetzt.}
\begin{itemize}
    \item Übergang, Abschnitt~\ref{sec:requirements_catalogue} leitet acht Anforderungen ab, die hier gegen das Ergebnis zu halten sind
    \item Die kurative Reservierung ist nach der Reaktionszeit differenziert, und der Basisfall setzt die Klasse von etwa zwei Minuten, womit A1 erfüllt ist
    \item Die Vergütung setzt an der vorgehaltenen Leistung und an der Bindungsdauer an, und ein Abruf wird gesondert vergütet, womit A4 erfüllt ist
    \item Die Präqualifikation prüft technische Eigenschaften und nicht die Art der Anlage, sodass der Zugang nach A5 technologieoffen bleibt
    \item Die Beschaffung ist netzknotenscharf entworfen, das Optimierungsmodell bildet das Netz aber nicht ab, sodass A3 im Entwurf erfüllt und im Ergebnis nicht geprüft ist
    \item Über diese vier Anforderungen entscheidet damit der Produktzuschnitt und nicht der ermittelte Preis
\end{itemize}

\noindent\textbf{Absatz 2, die Teilnahme nach A6 lohnt sich, und zwar messbar.}
\begin{itemize}
    \item Übergang, ob der kurative Marktmechanismus Anbieter findet, entscheidet A6
    \item Die Zahlung bei vollständiger Bindung beträgt 420,0\,\TsdEurMWa{} gegen einen Referenzerlös von 340,6\,\TsdEurMWa{}
    \item Der Gesamterlös der Bezugsanlage liegt damit um 23,3 Prozent über dem Referenzerlös, obwohl die übrige Vermarktung vollständig entfällt
    % EIGENSTAENDIGE ABLEITUNG, vom Verfasser zu pruefen. Die Summe der
    % stuendlichen Indifferenzpreise uebersteigt die entgangene Vermarktung,
    % weil jede Stunde gegen ihre eigene beste Verwendung bepreist wird,
    % waehrend die Verwendungen um dieselbe Leistung konkurrieren.
    \item Die Summe der stündlichen Indifferenzpreise übersteigt die entgangene Vermarktung, weil jede Stunde gegen ihre eigene beste Verwendung bepreist wird
    \item Ein Betreiber, der seine Preise richtig setzt, steigert seinen Erlös damit allein durch eine andere Verteilung auf die Märkte
    \item A6 ist erfüllt, und der Befund gilt für die Bezugsanlage und nicht für jede Anlage, denn der Preis folgt ihren Erlösmöglichkeiten
\end{itemize}

\noindent\textbf{Absatz 3, die Verträglichkeit nach A7 ist der kritische Punkt.}
\begin{itemize}
    \item Übergang, die schärfste der acht Anforderungen ist A7
    \item A7 verlangt, dass der kurative Marktmechanismus neben den bestehenden Märkten besteht, ohne einen von ihnen zu verdrängen
    \item Unter den Preisen der zweiten Iteration verdrängt die kurative Reservierung die gesamte übrige Vermarktung der Bezugsanlage
    \item Am stärksten trifft das die \ac{aFRR}-Leistung, auf die 65,6 Prozent des Bruttoerlöses entfallen
    \item Würde eine große Zahl geeigneter Anlagen kurativ gebunden, verlöre der Markt für \ac{aFRR}-Leistung seine Anbieter
    \item A7 ist damit nicht über den Produktzuschnitt zu erfüllen, sondern allein über eine Begrenzung der beschafften Menge
\end{itemize}

\noindent\textbf{Absatz 4, drei Anforderungen bleiben offen, und eine Konkurrenz kommt hinzu.}
\begin{itemize}
    \item Übergang, drei Anforderungen lassen sich aus dem Ergebnis nicht beantworten
    \item A2 Verbindlichkeit verlangt eine Sanktion, die das Optimierungsmodell nicht abbildet, sodass ihre Höhe offen bleibt
    \item A8 Umsetzbarkeit betrifft die internen Abläufe des \ac{ÜNB} und ist von außen nicht zu beurteilen
    \item Hinzu tritt eine Konkurrenz, die der Katalog nicht führt, denn die Präqualifikation der \ac{FCR} reserviert bereits ein Ladezustandsband \cite{ubertragungsnetzbetreiber_deutschland_praqualifikationsverfahren_2024}
    \item A5 sichert zudem den Zugang und nicht den Wettbewerb, denn an einem Knoten mit wenigen wirksamen Anlagen liegt das Gebot über dem ermittelten kurativen Reservierungspreis
    \item Für den kurativen Marktmechanismus folgt daraus, dass die Zahl der Anbieter je Netzknoten über die Lage des Preises entscheidet
\end{itemize}

\section{Folgen für die Betriebsführung und den Produktzuschnitt}
\label{sec:operational_consequences}
% Ziel 2 Seiten, vier Absaetze. Hier loesen sich G1 bis G4 auf, dazu D3, D4
% und die Vormerkungen aus 4.2 und 4.6.2.
% Fragen des Abschnitts: Was bedeutet das Ergebnis fuer den Betreiber, was
% fuer den UENB, und welcher Produktzuschnitt folgt daraus?

\noindent\textbf{Absatz 1, was der Betreiber aus dem Ergebnis macht.}
\begin{itemize}
    \item Übergang, der ermittelte Preis ist zuerst eine Anweisung an den Betreiber
    \item Der Betreiber muss je Stunde und Richtung ein Gebot bilden, das die Opportunität derselben Stunde trifft
    \item Den Leistungspreis der \ac{aFRR} kennt er dabei, denn die Ausschreibung schließt am Vortag um 9 Uhr \cite{regelleistung_ausschreibungsdaten_2026}
    \item Die Handelsspannen des folgenden Tages kennt er dagegen nicht, und gerade sie tragen die teuren Stunden
    \item Wer zu niedrig bietet, bindet seine Leistung unter Wert, und wer zu hoch bietet, erhält keinen Zuschlag
    \item Die kurative Reservierung gibt dem Betreiber damit eine Möglichkeit, seinen Erlös allein durch eine klügere Verteilung auf die Märkte zu steigern
\end{itemize}

\noindent\textbf{Absatz 2, die Granularität der Zeitscheibe bestimmt die Zahlung des \ac{ÜNB}.}
\begin{itemize}
    \item Übergang, für den \ac{ÜNB} entscheidet der Zuschnitt des Produkts über die Zahlung
    \item Die Preise der ersten Iteration geben allen Stunden eines Tages denselben Wert
    \item Sie binden damit 85,9 Prozent der Stunden entladend und 89,6 Prozent ladend und lassen gerade die teuren Stunden frei
    \item Der \ac{ÜNB} zahlt dabei 97,6 Prozent der Zahlung unter vollständiger Bindung
    \item Ein Produkt mit einer Zeitscheibe von einem Tag ist damit teuer und bindet das Falsche
    % Handlungsempfehlung, folgt aus 4.2.
    \item Die kurative Reservierung ist deshalb je Stunde auszuschreiben und nicht je Tag
\end{itemize}

\noindent\textbf{Absatz 3, die Abrufdauer ist die Stellschraube des \ac{ÜNB}.}
\begin{itemize}
    \item Übergang, über die Abrufdauer entscheidet allein der \ac{ÜNB}, denn sie folgt aus der Zeit, über die eine kurative Maßnahme wirken muss
    \item Die Abrufdauer setzt die Breite des reservierten Ladezustandsbandes
    \item Von einer Stunde auf eine Viertelstunde sinkt der Median um 4,0 Prozent entladend und um 12,7 Prozent ladend
    \item Die halbe Stunde liegt dabei fast auf dem Preis der Viertelstunde, nämlich bei 11,91 gegen 11,72\,\EurMWh{} entladend
    \item Eine halbe Stunde Abrufdauer verschafft dem \ac{ÜNB} die doppelte Zeit, die kurative Maßnahme durch eine andere abzulösen
    % Handlungsempfehlung, folgt aus 4.6.2.
    \item Für den Zuschnitt folgt daraus, die Abrufdauer auf eine halbe Stunde zu setzen
\end{itemize}

\noindent\textbf{Absatz 4, die Vergütung der Vorhaltung verschiebt das Anreizproblem.}
\begin{itemize}
    \item Übergang, mit dem Gegenstand der Vergütung verschieben sich die Anreize
    \item Hängt die Zahlung an der bereitgehaltenen Leistung, so entfällt der Anreiz, das Volumen des Eingriffs zu vergrößern
    \item Das Anreizproblem verschwindet damit nicht, sondern wandert
    \item An seine Stelle tritt der Anreiz, Leistung anzubieten, die nicht wirklich frei ist
    \item Aus der Frage nach der Menge des Eingriffs wird damit eine Frage des Nachweises
    % Handlungsempfehlung G2, folgt aus D4.
    \item Ein Verfahren zum Nachweis der Verfügbarkeit ist deshalb konstitutiver Bestandteil des kurativen Marktmechanismus und keine Frage der Ausführung
\end{itemize}

\section{Belastbarkeit und Grenzen}
\label{sec:robustness}
% Ziel 2,5 Seiten, fuenf Absaetze. Material 4.6, dazu F1 bis F4, F10, F11,
% E1 und E2 des alten Geruests.
% Fragen des Abschnitts: Welche Annahme traegt das Ergebnis? Wie wirkt die
% vollstaendige Preiskenntnis? Was laesst das Suchverfahren offen? Wie weit
% traegt eine Einzelanlagenbetrachtung? Was sagt die Verifikation?

\noindent\textbf{Absatz 1, die Modellierung der \ac{aFRR} trägt das Ergebnis.}
\begin{itemize}
    \item Übergang, die vier Sensitivitäten beantworten, woran das Ergebnis hängt
    \item Ohne die \ac{aFRR} fällt der Median auf 3,47 und 4,25\,\EurMWh{} und damit auf gut ein Viertel
    \item Unter freier Lieferung der \ac{aFRR}-Arbeit steigt er um 54 Prozent entladend und um 110 Prozent ladend
    \item Die Abrufdauer und die Handelsspanne am \ac{IDC} wirken demgegenüber schwächer
    \item Die anteilige Lieferung des Basisfalls ist die belastbare Annahme, denn eine Anlage von 100\,\si{MW} zieht den deutschlandweiten Abruf nicht auf sich
    \item Jede Aussage dieser Arbeit steht und fällt damit mit der Modellierung der \ac{aFRR}
\end{itemize}

\noindent\textbf{Absatz 2, die vollständige Preiskenntnis verzerrt in beide Richtungen.}
\begin{itemize}
    \item Übergang, die stärkste Annahme des Optimierungsmodells ist die vollständige Kenntnis der Preise
    \item Sie überschätzt die Opportunität, weil ein Betreiber die Preise des folgenden Tages nicht kennt
    \item Weber räumt dieselbe Verzerrung für sein eigenes Verfahren ein \cite{weber_gutachten_2015}
    \item Der Unterschied zwischen Erlöspotenzial und tatsächlich erzieltem Erlös ist unabhängig davon belegt \cite{mahgoub_wie_2025}
    \item Zugleich unterschlägt die vollständige Preiskenntnis den Optionswert der Bindung, denn bei bekannter Preisreihe ist er null
    \item Beide Verzerrungen laufen gegeneinander, sodass der ermittelte Preis als opportunitätskostenbasierter Wert zu lesen ist und nicht als vollständige Bewertung der Bindung
\end{itemize}

\noindent\textbf{Absatz 3, das Suchverfahren liefert ein koordinatenweises Minimum.}
% NICHT ALS FEHLER FORMULIEREN, sondern als Eigenschaft des Verfahrens.
\begin{itemize}
    \item Übergang, auch das Verfahren der Preissuche setzt dem Ergebnis eine Grenze
    \item Die Preise der zweiten Iteration sind ein koordinatenweises und kein globales Minimum
    \item Keine einzelne Stunde lässt sich um die Toleranz von 0,05\,\EurMWh{} senken, ohne dass die vollständige Bindung bricht
    \item Die Lage des Minimums hängt gleichwohl an der Reihenfolge des Abstiegs
    \item In umgekehrter Reihenfolge liegen die Preise am 11.02.2025 um 5,5 Prozent und am 26.08.2025 um 17,7 Prozent darunter, wobei beide Fassungen die Zusage tragen
    \item Der ausgewiesene Wert ist damit die vorsichtigere Seite einer gemessenen Spanne
\end{itemize}

\noindent\textbf{Absatz 4, eine einzelne Anlage trägt keine Aussage über das System.}
\begin{itemize}
    \item Übergang, die letzte Grenze ist die Perspektive der Untersuchung
    \item Diese Arbeit rechnet eine einzelne Anlage und bildet das Netz nicht ab
    \item Eine Hochrechnung auf die Systemebene setzte voraus, dass die Wirksamkeit eines Akteurs eine feste Eigenschaft seines Standorts ist
    \item Die Wirksamkeit ist keine feste Eigenschaft, denn der Redispatch aus Photovoltaik stieg im Jahr 2025 um 94 Prozent, während der Anteil der Offshore-Windenergie um 27 Prozent fiel \cite{bundesnetzagentur_smard_engpassmanagement_2026}
    \item Zugleich bildet das Optimierungsmodell weder eine Bietfunktion noch den Wettbewerb mehrerer Anbieter ab
    \item Die Aussagen über die Lage des Gebots im Wettbewerb bleiben deshalb qualitativ
\end{itemize}

\noindent\textbf{Absatz 5, was die Verifikation trägt und was nicht.}
\begin{itemize}
    \item Übergang, die Güte des Optimierungsmodells ist an einem externen Erlösindex geprüft
    \item Der Vergleich stellt ein Modell mit vollständiger Preiskenntnis einem Verfahren mit rollierender Vorausschau gegenüber
    \item Er stellt zudem einen engeren Marktumfang einem weiteren gegenüber
    \item Beide Unterschiede wirken in dieselbe Richtung und erklären einen Teil der Abweichung
    \item Die Verifikation belegt damit die Größenordnung der Erlöse und nicht die Höhe einer einzelnen Stunde
\end{itemize}

\noindent\textbf{Absatz 6, was die Sensitivität zur Handelsspanne nicht misst.}
% VORGEMERKT am 24.09.2026, Vorgabe des Verfassers: zur Wahrheit gehoert,
% dass die Sensitivitaet in 4.6.2 nicht misst, was sie messen soll. Der
% Faktor streckt um das Tagesmittel, der Vorteil des mehrfachen Handels

% Handel derselben Viertelstunde, wie in Abschnitt 3.2 eingefuehrt.
\begin{itemize}
    \item Übergang, auch eine Sensitivität kann die Frage verfehlen, die sie beantworten soll
    \item Der Faktor auf die Handelsspanne streckt jede Abweichung des Intradaypreises von seinem Tagesmittel
    \item Er hebt damit gerade die Stunden, die ohnehin weit vom Tagesmittel liegen, also die schon teuren und die schon günstigen
    \item Der Vorteil des mehrfachen Handels entsteht dagegen in Viertelstunden mit großer Preisbewegung innerhalb desselben Lieferzeitpunkts
    \item Beide Mengen von Stunden fallen nicht zusammen, sodass die Sensitivität allein die Breite der Handelsspanne misst

\end{itemize}

    \item Niedrigere Marktpreise machen die kurative Bindung damit nicht günstiger als der heutige Stand

\end{itemize}

\section{Zukunft der \acs{BESS} im Engpassmanagement}
\label{sec:bess_outlook}
% Ziel 1,5 Seiten, drei Absaetze. Der Abschnitt, den der Verfasser am
% 23.09.2026 ausdruecklich verlangt hat.
% Fragen des Abschnitts: Warum ist ein BESS fuer die kurative Vorhaltung
% geeignet? Wo liegt seine Grenze? Wie veraendert sich das mit dem Bestand?

\noindent\textbf{Absatz 1, warum ein \ac{BESS} für die kurative Vorhaltung geeignet ist.}
\begin{itemize}
    \item Übergang, unter den Technologien ist das \ac{BESS} für die kurative Vorhaltung besonders geeignet
    \item Es erreicht die Reaktionszeitklasse von etwa zwei Minuten ohne Anfahrvorgang \cite{innosys_2030_gesamtverbund_innovationen_2021}
    \item Es stellt beide Richtungen aus derselben Anlage, sodass eine bilanziell ausgeglichene Vorhaltung an einem Netzknoten möglich ist
    \item Ein Energieinhalt je Leistung von einer bis zwei Stunden genügt für eine Abrufdauer von einer halben Stunde
    \item Der Bestand wächst, und die für die \ac{FCR} präqualifizierte Leistung von 810\,\si{MW} im Jahr 2024 zeigt, wie schnell \ac{BESS} einen Regelleistungsmarkt füllen
    \item Die kurative Vorhaltung trifft damit auf eine Technologie, deren Bestand ohnehin zunimmt
\end{itemize}

\noindent\textbf{Absatz 2, wo die Grenze der Eignung liegt.}
\begin{itemize}
    \item Übergang, die Eignung hat drei Grenzen
    \item Die erste ist der Energieinhalt, denn eine längere Abrufdauer verlangt ein breiteres Ladezustandsband und verteuert die Vorhaltung
    \item Die zweite ist die Doppelbelegung dieses Bandes, das die Vorhaltung von Regelleistung bereits beansprucht
    \item Die dritte ist die örtliche Wirksamkeit, die nicht an der Technologie hängt, sondern am Netzknoten
    \item Ein \ac{BESS} am falschen Netzknoten ist für die Engpassbehebung wertlos, so schnell es auch reagiert
    \item Über den Nutzen entscheidet damit die Standortwahl ebenso wie die technische Eignung
\end{itemize}

\noindent\textbf{Absatz 3, wie sich der Einsatz mit dem Bestand verschiebt.}
% Vorgabe des Verfassers vom 23.09.2026, HANDOFF Abschnitt 7.
\begin{itemize}
    \item Übergang, mit dem Bestand verschiebt sich, wofür ein \ac{BESS} eingeplant wird
    \item Ein \ac{BESS} wird eher zum Entladen als bilanzieller Ausgleich für erneuerbare Erzeugung eingeplant
    \item Gekoppelt mit regelbarer Photovoltaik lohnen deshalb die Mittagsstunden am meisten
    \item Saisonal liegt der größte Nutzen im Winter, in dem die Winderzeugung überwiegt und der Engpassmanagementbedarf am größten ist
    \item Die Abendstunden sind dagegen teuer und treffen keinen erhöhten Bedarf
    \item Für die Beschaffung folgt daraus, die kurative Vorhaltung dort zu beschaffen, wo ein hoher Bedarf auf günstige Stunden trifft
\end{itemize}

\section{Kurative Systemführung im Allgemeinen}
\label{sec:curative_general}
% Ziel 1,5 Seiten, drei Absaetze. Der Abschluss, den der Verfasser am
% 23.09.2026 verlangt hat: erst die Verallgemeinerung, daraus der
% Forschungsbedarf und der Erkenntnisbedarf aus Projekten.

\noindent\textbf{Absatz 1, was jenseits des Speichers gilt.}
\begin{itemize}
    \item Übergang, die Befunde hängen an der Bezugsanlage, der kurative Marktmechanismus hängt nicht an ihr
    \item Der kurative Reservierungspreis ist für jede Technologie die entgangene Marktopportunität der gebundenen Stunde
    \item Für ein Kraftwerk, das ohnehin am Netz ist, ist diese Opportunität eine andere als für einen Speicher
    \item Der kurative Marktmechanismus bleibt damit technologieoffen, und die Höhe des Gebots unterscheidet die Technologien
    \item Der Vergleich mehrerer Technologien an einem Netzknoten ist die eigentliche Aufgabe der Beschaffung
    \item Diese Arbeit liefert dafür den Preis einer Technologie und nicht die Rangfolge
\end{itemize}

\noindent\textbf{Absatz 2, welcher Forschungsbedarf offen bleibt.}
\begin{itemize}
    \item Übergang, aus den Grenzen dieser Arbeit folgt der weitere Forschungsbedarf
    \item Die Wahrscheinlichkeit eines Abrufs und damit der Optionswert der Bindung sind zu bestimmen
    \item Eine Hochrechnung auf die Systemebene verlangt ein Modell, das das Netz und die Wirksamkeit je Netzknoten führt
    \item Das Bietverhalten und der Wettbewerb mehrerer Anbieter an einem Netzknoten sind abzubilden
    \item Die Höhe einer Sanktion, die eine verspätete Leistungsänderung wirksam ahndet, ist zu bestimmen
    \item Und der Vergleich der Technologien verlangt Opportunitätskosten auch für Kraftwerke und für Lasten
\end{itemize}

\noindent\textbf{Absatz 3, welche Erkenntnisse aus Projekten zu gewinnen sind.}
\begin{itemize}
    \item Übergang, einen Teil dieser Fragen können allein Projekte im Echtbetrieb beantworten
    \item InnoSys 2030 hat die kurative Systemführung erprobt und die Reaktionszeitklassen geliefert \cite{innosys_2030_gesamtverbund_innovationen_2021}
    \item KuPilot erprobt die kurative Höherauslastung im realen Betrieb \cite{tennet_tso_gmbh_pilotbetrieb_2025}
    \item Offen bleibt, wie häufig und wie lange eine kurative Maßnahme tatsächlich abgerufen wird
    \item Erst daraus folgt, ob die gesonderte Vergütung des Abrufs neben der Vergütung der Vorhaltung ins Gewicht fällt
    \item Ebenso offen ist, wie sich die Verfügbarkeit einer zugesagten Leistung im Betrieb nachweisen lässt, und beide Fragen entscheiden über die Umsetzbarkeit des kurativen Marktmechanismus
\end{itemize}
