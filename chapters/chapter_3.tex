\subsection{Intraday Spotmarkt}
\label{sec:market_id}

\begin{itemize}
    \item Kontinuierlicher Handel und Intraday-Auktionen (ID1, ID3) für Stunden- und Viertelstundenprodukte
    \item Gate Closure 30 Minuten vor Lieferbeginn in Deutschland
    \item Hohe Relevanz für kurativen Einsatz: kurzfristige Netzengpässe erst im Intraday-Horizont sichtbar
    \item Viertelstundenprodukte als geeignetes Instrument für schnelle Batteriespeicher
\end{itemize}

\label{sec:market_fcr}

\begin{itemize}
    \item Automatische Frequenzstabilisierung innerhalb von 30 Sekunden (FCR)
    \item Symmetrisches Produkt: gleichzeitige Vorhaltung von positiver und negativer Leistung

\end{itemize}

\subsubsection{Sekundärregelleistung}
\label{sec:market_afrr}

\begin{itemize}
    \item Automatisch aktivierte Frequenzwiederherstellung (aFRR), Reaktionszeit 30 Sekunden
    \item Getrennte Ausschreibung von positiver und negativer Regelleistung
    \item Aktivierungswahrscheinlichkeit und Erlösstruktur für Speicherbetreiber
\end{itemize}

\subsubsection{Minutenreserve}
\label{sec:market_mfrr}

\begin{itemize}

    \item Geringere Anforderungen an Reaktionsgeschwindigkeit, aber höhere Aktivierungswahrscheinlichkeit
\end{itemize}

\subsubsection{Präventiver Redispatch und kurative Höherauslastung}
\label{sec:redispatch_prev_cur}

\begin{itemize}
    \item Präventiver Redispatch: Anpassung von Einspeisefahrplänen vor Eintreten des Engpasses im Day-Ahead- oder Intraday-Horizont
    \item Kurativer Redispatch: Eingriff nach Eintreten des Engpasses durch Anpassung von Wirkleistungseinspeisung in Echtzeit
    \item Kurative Höherauslastung: Nutzung der transienten Stromtragfähigkeit (TATL) zur temporären Überschreitung des Dauergrenzwerts (PATL) - kein Redispatch, sondern Erweiterung des zulässigen Betriebsbereichs
    \item Abgrenzung: Kurative Höherauslastung ersetzt oder reduziert den Bedarf an kurativem Redispatch, indem der Engpass durch Tolerierung transienter Überlastung überbrückt wird
    \item Effizienzpotenzial: Kurative Höherauslastung ermöglicht Kosteneinsparung gegenüber präventivem Redispatch bei gleichzeitiger Wahrung der Netzsicherheit
\end{itemize}
